import { Router } from 'express'
import named from '../../../utils/namedRoutes'
import UserController from '../http/controllers/api/v1/UserController'
import PermissionController from '../http/controllers/api/v1/PermissionController'
import AccessMiddleware from '../http/middleware/AccessMiddleware'
import { ensureAuthenticatedApi } from '../../auth/http/middleware/authMiddleware'
import RoleController from '../http/controllers/api/v1/RoleController'
import { roleValidationRules } from '../http/validators/RoleValidator'
import { permissionValidationRules } from '../http/validators/PermissionValidator'
import { upload, UserCreateValidator } from '../http/validators/UserCreateValidator'
import { UserUpdateValidator } from '../http/validators/UserUpdateValidator'
 
const router = Router()
 
// define route & set middleware user
const userRoute = named.extendRouter(Router())
 
const userController = new UserController()
userRoute.get('api.v1.access.user.index', '/api/v1/access/user', ensureAuthenticatedApi, AccessMiddleware, userController.index.bind(userController))
userRoute.post('api.v1.access.user.store', '/api/v1/access/user/store', ensureAuthenticatedApi, AccessMiddleware, upload.any(), UserCreateValidator, userController.store.bind(userController))
userRoute.get('api.v1.access.user.edit', '/api/v1/access/user/:id/edit', ensureAuthenticatedApi, AccessMiddleware, userController.edit.bind(userController))
userRoute.put('api.v1.access.user.update', '/api/v1/access/user/:id/update', ensureAuthenticatedApi, AccessMiddleware, upload.any(), UserUpdateValidator, userController.update.bind(userController))
userRoute.get('api.v1.access.user.delete', '/api/v1/access/user/:id/delete', ensureAuthenticatedApi, AccessMiddleware, userController.delete.bind(userController))
userRoute.post('api.v1.access.user.delete_selected', '/api/v1/access/user/delete_selected', ensureAuthenticatedApi, AccessMiddleware, userController.delete_selected.bind(userController))
 
// define route & set middleware access
const permissionRoute = named.extendRouter(Router())
 
const permissionController = new PermissionController()
permissionRoute.get('api.v1.access.permission.index', '/api/v1/access/permission', ensureAuthenticatedApi, AccessMiddleware, permissionController.index.bind(permissionController))
permissionRoute.post('api.v1.access.permission.store', '/api/v1/access/permission/store', ensureAuthenticatedApi, AccessMiddleware, permissionValidationRules(), permissionController.store.bind(permissionController))
permissionRoute.get('api.v1.access.permission.edit', '/api/v1/access/permission/:id/edit', ensureAuthenticatedApi, AccessMiddleware, permissionController.edit.bind(permissionController))
permissionRoute.put('api.v1.access.permission.update', '/api/v1/access/permission/:id/update', ensureAuthenticatedApi, AccessMiddleware, permissionValidationRules(), permissionController.update.bind(permissionController))
permissionRoute.get('api.v1.access.permission.delete', '/api/v1/access/permission/:id/delete', ensureAuthenticatedApi, AccessMiddleware, permissionController.delete.bind(permissionController))
permissionRoute.post('api.v1.access.permission.delete_selected', '/api/v1/access/permission/delete_selected', ensureAuthenticatedApi, AccessMiddleware, permissionController.delete_selected.bind(permissionController))
 
// define route & set middleware role
const roleRoute = named.extendRouter(Router())
 
const roleController = new RoleController()
roleRoute.get('api.v1.access.role.index', '/api/v1/access/role', ensureAuthenticatedApi, AccessMiddleware, roleController.index.bind(roleController))
roleRoute.post('api.v1.access.role.store', '/api/v1/access/role/store', ensureAuthenticatedApi, AccessMiddleware, roleValidationRules(), roleController.store.bind(roleController))
roleRoute.get('api.v1.access.role.permission', '/api/v1/access/role/:id/permission', ensureAuthenticatedApi, AccessMiddleware, roleController.permission.bind(roleController))
roleRoute.get('api.v1.access.role.permission.assign', '/api/v1/access/role/:id/permission/:permission_id/assign', ensureAuthenticatedApi, AccessMiddleware, roleController.permission_assign.bind(roleController))
roleRoute.post('api.v1.access.role.permission.assign_selected', '/api/v1/access/role/:id/permission/assign_selected', ensureAuthenticatedApi, AccessMiddleware, roleController.permission_assign_selected.bind(roleController))
roleRoute.get('api.v1.access.role.permission.unassign', '/api/v1/access/role/:id/permission/:permission_id/unassign', ensureAuthenticatedApi, AccessMiddleware, roleController.permission_unassign.bind(roleController))
roleRoute.post('api.v1.access.role.permission.unassign_selected', '/api/v1/access/role/:id/permission/unassign_selected', ensureAuthenticatedApi, AccessMiddleware, roleController.permission_unassign_selected.bind(roleController))
roleRoute.get('api.v1.access.role.edit', '/api/v1/access/role/:id/edit', ensureAuthenticatedApi, AccessMiddleware, roleController.edit.bind(roleController))
roleRoute.put('api.v1.access.role.update', '/api/v1/access/role/:id/update', ensureAuthenticatedApi, AccessMiddleware, roleValidationRules(), roleController.update.bind(roleController))
roleRoute.get('api.v1.access.role.delete', '/api/v1/access/role/:id/delete', ensureAuthenticatedApi, AccessMiddleware, roleController.delete.bind(roleController))
roleRoute.post('api.v1.access.role.delete_selected', '/api/v1/access/role/delete_selected', ensureAuthenticatedApi, AccessMiddleware, roleController.delete_selected.bind(roleController))
 
router.use(userRoute,permissionRoute,roleRoute)
 
export default router
